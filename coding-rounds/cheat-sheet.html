Two Pointers

O(n)

When to Use

Sorted arrays, finding pairs, palindromes, container problems

Two Sum (sorted) Container With Most Water Valid Palindrome 3Sum

Template

Python

def two_pointers(arr):
    left, right = 0, len(arr) - 1
    while left < right:
        if condition_met:
            return result
        elif need_bigger:
            left += 1
        else:
            right -= 1

Java

public int twoPointers(int[] arr) {
    int left = 0, right = arr.length - 1;
    while (left < right) {
        if (conditionMet(arr, left, right))
            return result;
        else if (needBigger)
            left++;
        else
            right--;
    }
    return -1;
}
Key: Works on sorted arrays. Move pointers based on comparison with target.

Sliding Window

O(n)

When to Use

Subarray/substring problems, contiguous sequences, max/min in window

Max Sum Subarray Longest Substring Without Repeats Minimum Window Substring Find All Anagrams

Template (Variable Size)

Python

def sliding_window(s):
    left = 0
    window = {}  # or other state
    result = 0

    for right in range(len(s)):
        # Expand window
        window[s[right]] = window.get(s[right], 0) + 1

        # Contract window while invalid
        while window_invalid():
            window[s[left]] -= 1
            left += 1

        # Update result
        result = max(result, right - left + 1)

    return result

Java

public int slidingWindow(String s) {
    Map<Character, Integer> window = new HashMap<>();
    int left = 0, result = 0;
    for (int right = 0; right < s.length(); right++) {
        char c = s.charAt(right);
        window.merge(c, 1, Integer::sum);
        while (windowInvalid(window)) {
            char l = s.charAt(left++);
            window.merge(l, -1, Integer::sum);
            if (window.get(l) == 0) window.remove(l);
        }
        result = Math.max(result, right - left + 1);
    }
    return result;
}
Key: Expand right, contract left when constraint violated. Track window state with hashmap.

Binary Search

O(log n)

When to Use

Sorted arrays, search space reduction, finding boundaries

Search in Rotated Array Find First/Last Position Search Insert Position Koko Eating Bananas

Template (Find Leftmost)

Python

def binary_search_left(arr, target):
    left, right = 0, len(arr)
    while left < right:
        mid = (left + right) // 2
        if arr[mid] < target:
            left = mid + 1
        else:
            right = mid
    return left  # First position >= target

Java

public int binarySearchLeft(int[] arr, int target) {
    int left = 0, right = arr.length;
    while (left < right) {
        int mid = left + (right - left) / 2;
        if (arr[mid] < target)
            left = mid + 1;
        else
            right = mid;
    }
    return left; // first position >= target
}
Key: Use left < right with right = mid for leftmost. For rightmost, use left <= right with left = mid + 1 and track the last match. Use mid = left + (right - left) / 2 in Java to avoid overflow.

BFS (Breadth-First Search)

O(V + E)

When to Use

Shortest path (unweighted), level-order traversal, nearest neighbor

Level Order Traversal Word Ladder Rotting Oranges Shortest Path in Grid

Template

Python

from collections import deque

def bfs(graph, start):
    visited = {start}
    queue = deque([start])

    while queue:
        node = queue.popleft()
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

Java

public void bfs(Map<Integer, List<Integer>> graph, int start) {
    Set<Integer> visited = new HashSet<>();
    Deque<Integer> queue = new ArrayDeque<>();
    visited.add(start);
    queue.offer(start);
    while (!queue.isEmpty()) {
        int node = queue.poll();
        for (int neighbor : graph.getOrDefault(node, List.of())) {
            if (visited.add(neighbor))
                queue.offer(neighbor);
        }
    }
}
Key: Use deque for O(1) popleft. Add to visited BEFORE adding to queue to avoid duplicates.

DFS (Depth-First Search)

O(V + E)

When to Use

Path finding, cycle detection, connected components, tree traversal

Number of Islands Path Sum Clone Graph Course Schedule

Template (Recursive)

Python

def dfs(graph, node, visited):
    if node in visited:
        return
    visited.add(node)

    for neighbor in graph[node]:
        dfs(graph, neighbor, visited)

Java

public void dfs(Map<Integer, List<Integer>> graph,
                int node, Set<Integer> visited) {
    if (!visited.add(node)) return;
    for (int neighbor : graph.getOrDefault(node, List.of()))
        dfs(graph, neighbor, visited);
}
Key: Recursive = implicit stack. For large inputs, use iterative with explicit stack to avoid stack overflow.

Backtracking

O(n! or 2^n)

When to Use

Permutations, combinations, subsets, constraint satisfaction

Subsets Permutations Combination Sum N-Queens

Template

Python

def backtrack(path, choices):
    if is_solution(path):
        result.append(path[:])
        return

    for choice in choices:
        if is_valid(choice):
            path.append(choice)      # Make choice
            backtrack(path, remaining_choices)
            path.pop()               # Undo choice

Java

private void backtrack(List<Integer> path,
                        int[] choices, List<List<Integer>> result) {
    if (isSolution(path)) {
        result.add(new ArrayList<>(path));
        return;
    }
    for (int choice : choices) {
        if (isValid(choice, path)) {
            path.add(choice);                    // make choice
            backtrack(path, choices, result);
            path.remove(path.size() - 1);        // undo choice
        }
    }
}
Key: Always undo your choice after recursing. Use path[:] / new ArrayList<>(path) to copy the path when saving.

Dynamic Programming

O(n) to O(n²)

When to Use

Optimization problems, counting problems, overlapping subproblems

Climbing Stairs Coin Change Longest Common Subsequence Edit Distance

Template (Bottom-Up)

Python

def dp_solution(n):
    # 1. Define state: dp[i] = answer for subproblem i
    dp = [0] * (n + 1)

    # 2. Base cases
    dp[0] = base_value

    # 3. Fill table using recurrence
    for i in range(1, n + 1):
        dp[i] = recurrence(dp[i-1], ...)

    return dp[n]

Java

public int dpSolution(int n) {
    // 1. Define state: dp[i] = answer for subproblem i
    int[] dp = new int[n + 1];

    // 2. Base cases
    dp[0] = baseValue;

    // 3. Fill table using recurrence
    for (int i = 1; i <= n; i++)
        dp[i] = recurrence(dp[i - 1]);

    return dp[n];
}
Key: Define state clearly first. Identify base cases. Write recurrence relation. Consider space optimization.

Topological Sort

O(V + E)

When to Use

Dependency ordering, course scheduling, build systems, DAG problems

Course Schedule Alien Dictionary Build System Dependencies

Template (Kahn's Algorithm)

Python

def topological_sort(n, edges):
    in_degree = [0] * n
    graph = defaultdict(list)

    for u, v in edges:
        graph[u].append(v)
        in_degree[v] += 1

    queue = deque([i for i in range(n) if in_degree[i] == 0])
    result = []

    while queue:
        node = queue.popleft()
        result.append(node)
        for neighbor in graph[node]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    return result if len(result) == n else []  # Empty = cycle

Java

public int[] topologicalSort(int n, int[][] edges) {
    int[] inDegree = new int[n];
    List<List<Integer>> graph = new ArrayList<>();
    for (int i = 0; i < n; i++) graph.add(new ArrayList<>());
    for (int[] e : edges) { graph.get(e[0]).add(e[1]); inDegree[e[1]]++; }

    Deque<Integer> queue = new ArrayDeque<>();
    for (int i = 0; i < n; i++) if (inDegree[i] == 0) queue.offer(i);

    int[] result = new int[n];
    int idx = 0;
    while (!queue.isEmpty()) {
        int node = queue.poll();
        result[idx++] = node;
        for (int nb : graph.get(node))
            if (--inDegree[nb] == 0) queue.offer(nb);
    }
    return idx == n ? result : new int[0]; // empty = cycle
}
Key: If result length != n, there's a cycle. Start with nodes having in_degree 0.

Union-Find (Disjoint Set)

O(α(n)) ≈ O(1)

When to Use

Connected components, cycle detection, dynamic connectivity

Number of Connected Components Redundant Connection Accounts Merge

Template

Python

class UnionFind:
    def __init__(self, n):
        self.parent = list(range(n))
        self.rank = [0] * n

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])  # Path compression
        return self.parent[x]

    def union(self, x, y):
        px, py = self.find(x), self.find(y)
        if px == py:
            return False
        if self.rank[px] < self.rank[py]:
            px, py = py, px
        self.parent[py] = px
        if self.rank[px] == self.rank[py]:
            self.rank[px] += 1
        return True

Java

class UnionFind {
    int[] parent, rank;
    UnionFind(int n) {
        parent = new int[n]; rank = new int[n];
        for (int i = 0; i < n; i++) parent[i] = i;
    }
    int find(int x) {
        if (parent[x] != x) parent[x] = find(parent[x]); // path compression
        return parent[x];
    }
    boolean union(int x, int y) {
        int px = find(x), py = find(y);
        if (px == py) return false;
        if (rank[px] < rank[py]) { int t = px; px = py; py = t; }
        parent[py] = px;
        if (rank[px] == rank[py]) rank[px]++;
        return true;
    }
}
Key: Path compression + union by rank = nearly O(1). Union returns False if already connected (cycle detection).

Heap / Priority Queue

O(log n) per operation

When to Use

Top K elements, merge K sorted lists, streaming median

Kth Largest Element Merge K Sorted Lists Find Median from Stream Meeting Rooms II

Template

Python

import heapq

# Min heap (default in Python)
heap = []
heapq.heappush(heap, value)
min_val = heapq.heappop(heap)

# Max heap: negate values
heapq.heappush(heap, -value)
max_val = -heapq.heappop(heap)

# Kth largest: maintain min heap of size k
def kth_largest(nums, k):
    heap = nums[:k]
    heapq.heapify(heap)
    for num in nums[k:]:
        if num > heap[0]:
            heapq.heapreplace(heap, num)
    return heap[0]

Java

// Min heap
PriorityQueue<Integer> minHeap = new PriorityQueue<>();

// Max heap
PriorityQueue<Integer> maxHeap = new PriorityQueue<>(Collections.reverseOrder());

// Kth largest: min heap of size k
public int kthLargest(int[] nums, int k) {
    PriorityQueue<Integer> heap = new PriorityQueue<>();
    for (int num : nums) {
        heap.offer(num);
        if (heap.size() > k) heap.poll();
    }
    return heap.peek();
}
Key: Python heapq is min heap. For max heap, negate values. For Kth largest, use min heap of size K.

Trie (Prefix Tree)

O(m) per operation

When to Use

Prefix matching, autocomplete, word search, dictionary operations

Implement Trie Word Search II Design Search Autocomplete

Template

Python

class TrieNode:
    def __init__(self):
        self.children = {}
        self.is_end = False

class Trie:
    def __init__(self):
        self.root = TrieNode()

    def insert(self, word):
        node = self.root
        for char in word:
            if char not in node.children:
                node.children[char] = TrieNode()
            node = node.children[char]
        node.is_end = True

    def search(self, word):
        node = self._find(word)
        return node is not None and node.is_end

    def starts_with(self, prefix):
        return self._find(prefix) is not None

    def _find(self, prefix):
        node = self.root
        for char in prefix:
            if char not in node.children:
                return None
            node = node.children[char]
        return node

Java

class Trie {
    private static class TrieNode {
        Map<Character, TrieNode> children = new HashMap<>();
        boolean isEnd = false;
    }
    private final TrieNode root = new TrieNode();

    public void insert(String word) {
        TrieNode node = root;
        for (char c : word.toCharArray()) {
            node.children.putIfAbsent(c, new TrieNode());
            node = node.children.get(c);
        }
        node.isEnd = true;
    }

    public boolean search(String word) {
        TrieNode node = find(word);
        return node != null && node.isEnd;
    }

    public boolean startsWith(String prefix) { return find(prefix) != null; }

    private TrieNode find(String s) {
        TrieNode node = root;
        for (char c : s.toCharArray()) {
            node = node.children.get(c);
            if (node == null) return null;
        }
        return node;
    }
}
Key: m = word length. Space: O(total characters inserted) with hashmap children; O(ALPHABET_SIZE * total nodes) with fixed-size arrays.

Monotonic Stack

O(n)

When to Use

Next greater/smaller element, histogram problems, temperature spans

Daily Temperatures (LC 739) Largest Rectangle in Histogram (LC 84) Trapping Rain Water (LC 42) Next Greater Element (LC 496)

Template

Python

def next_greater(nums):
    n = len(nums)
    result = [-1] * n
    stack = []  # indices, decreasing values
    for i in range(n):
        while stack and nums[stack[-1]] < nums[i]:
            result[stack.pop()] = nums[i]
        stack.append(i)
    return result

Java

public int[] nextGreater(int[] nums) {
    int n = nums.length;
    int[] result = new int[n];
    Arrays.fill(result, -1);
    Deque<Integer> stack = new ArrayDeque<>();
    for (int i = 0; i < n; i++) {
        while (!stack.isEmpty() && nums[stack.peek()] < nums[i])
            result[stack.pop()] = nums[i];
        stack.push(i);
    }
    return result;
}
Key: Each element pushed/popped at most once → O(n) amortized. Stack maintains decreasing order of values (for next-greater variant).

Greedy

O(n log n) typical

When to Use

Interval scheduling, jump games, activity selection, when locally optimal = globally optimal

Merge Intervals (LC 56) Jump Game (LC 55) Gas Station (LC 134) Task Scheduler (LC 621)

Template — Exchange Argument Pattern

Python

def greedy_intervals(intervals):
    intervals.sort(key=lambda x: x[1])  # sort by end
    count, prev_end = 0, float('-inf')
    for start, end in intervals:
        if start >= prev_end:
            count += 1
            prev_end = end
    return count

Java

public int greedyIntervals(int[][] intervals) {
    Arrays.sort(intervals, (a, b) -> a[1] - b[1]);
    int count = 0, prevEnd = Integer.MIN_VALUE;
    for (int[] iv : intervals) {
        if (iv[0] >= prevEnd) {
            count++;
            prevEnd = iv[1];
        }
    }
    return count;
}
Key: Prove correctness via exchange argument. If you can't find a counterexample in 30 seconds, code it. Sort by end time for most interval problems.

Bit Manipulation

O(n) or O(1)

When to Use

XOR tricks, power-of-2 checks, subset enumeration, find unique element

Single Number (LC 136) Missing Number (LC 268) Power of Two (LC 231) Subsets (LC 78)

Template

Python

# XOR cancellation: find unique element
def single_number(nums):
    result = 0
    for n in nums:
        result ^= n
    return result

# Essential tricks
# n & (n-1)  --> clear lowest set bit
# n & (-n)   --> isolate lowest set bit
# n & 1      --> check odd/even

Java

public int singleNumber(int[] nums) {
    int result = 0;
    for (int n : nums) result ^= n;
    return result;
}

// Essential tricks
// (n & (n-1)) == 0  --> n is a power of two (n > 0)
// n & (-n)          --> isolate lowest set bit
// (n & 1) == 1      --> n is odd
Key: XOR is self-inverse (a^a=0). n & (n-1) clears lowest set bit. n & (-n) isolates it. Enumerate all 2n subsets with a bitmask loop.

Divide & Conquer

O(n log n)

When to Use

Merge sort, quick select, closest pair, count inversions

Merge Sort Quick Select (LC 215) Count Inversions Maximum Subarray (LC 53)

Template

Python

def merge_sort(arr):
    if len(arr) <= 1:
        return arr
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    return merge(left, right)

def merge(left, right):
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i]); i += 1
        else:
            result.append(right[j]); j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

Java

public void mergeSort(int[] arr, int lo, int hi) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    mergeSort(arr, lo, mid);
    mergeSort(arr, mid + 1, hi);
    merge(arr, lo, mid, hi);
}

private void merge(int[] arr, int lo, int mid, int hi) {
    int[] tmp = Arrays.copyOfRange(arr, lo, hi + 1);
    int i = 0, j = mid - lo + 1, k = lo;
    while (i <= mid - lo && j <= hi - lo)
        arr[k++] = (tmp[i] <= tmp[j]) ? tmp[i++] : tmp[j++];
    while (i <= mid - lo) arr[k++] = tmp[i++];
    while (j <= hi - lo) arr[k++] = tmp[j++];
}
Key: T(n) = 2T(n/2) + O(n) → O(n log n) by Master Theorem case 2. Count inversions by augmenting the merge step.

Segment Tree / BIT (Fenwick Tree)

O(log n) per query/update

When to Use

Range sum/min/max queries with updates, mutable range problems

Range Sum Query Mutable (LC 307) Count Smaller After Self (LC 315) Reverse Pairs (LC 493)

Template — Fenwick Tree

Python

class FenwickTree:
    def __init__(self, n):
        self.n = n
        self.tree = [0] * (n + 1)

    def update(self, i, delta):
        while i <= self.n:
            self.tree[i] += delta
            i += i & (-i)

    def query(self, i):
        s = 0
        while i > 0:
            s += self.tree[i]
            i -= i & (-i)
        return s

Java

class FenwickTree {
    int[] tree;
    int n;
    FenwickTree(int n) { this.n = n; tree = new int[n + 1]; }

    void update(int i, int delta) {
        for (; i <= n; i += i & (-i)) tree[i] += delta;
    }

    int query(int i) {
        int s = 0;
        for (; i > 0; i -= i & (-i)) s += tree[i];
        return s;
    }

    // Range sum [l, r] (1-indexed)
    int rangeQuery(int l, int r) { return query(r) - query(l - 1); }
}
Key: Fenwick for prefix sums (simpler, faster constant). Segment tree for range min/max or lazy propagation. i & (-i) isolates the lowest set bit — the core trick.
